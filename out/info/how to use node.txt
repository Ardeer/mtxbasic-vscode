const fs = require('fs');
const path = require('path');
const { basicTokeniseLine } = require('../out/mtxtokens.js');

const filePath = path.join(__dirname, '..', 'out', 'PANELCOD.BAS');
if (!fs.existsSync(filePath)) {
  console.error('File not found:', filePath);
  process.exit(2);
}

const txt = fs.readFileSync(filePath, 'utf8');
const lines = txt.split(/\r?\n/);

console.log('Tokenising file:', filePath);
for (const line of lines) {
  if (!line.trim()) continue;
  try {
    const bytes = basicTokeniseLine(line);
    console.log('LINE:', line);
    console.log('BYTES:', bytes);
  } catch (err) {
    console.error('Error tokenising line:', line, err && err.stack ? err.stack : err);
  }
}

console.log('Done.');
